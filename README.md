            const filteredContainer = document.getElementById('filteredResultContainer');
            const filteredGrid = document.getElementById('filteredGrid');
            const filteredTitle = document.getElementById('filteredTitle');

            if (!q) {
                filteredContainer.classList.add('hidden');
                return;

                p.categoryName.toLowerCase().includes(q) ||

            );

            filteredTitle.textContent = `KẾT QUẢ TÌM KIẾM "${keyword}" (${matched.length} SẢN PHẨM)`;

        /* Điều khiển 2 Banner Tự Trượt Chuyên Nghiệp */
        let currentSlideIndex = 0;
        let slideInterval = null;

        function showSlide(index) {
            const slides = document.querySelectorAll('.hero-slide');
            const dots = document.querySelectorAll('.slide-dot');
            if (!slides.length) return;

            currentSlideIndex = (index + slides.length) % slides.length;
            slides.forEach((sl, i) => {
                if (i === currentSlideIndex) {
                    sl.classList.remove('hidden', 'opacity-0');
                    sl.classList.add('block', 'opacity-100');
                } else {
                    sl.classList.remove('block', 'opacity-100');
                    sl.classList.add('hidden', 'opacity-0');
                }
            });

            dots.forEach((d, i) => {
                if (i === currentSlideIndex) {
                    d.classList.add('bg-amber-400');
                    d.classList.remove('bg-white/50');
                } else {
                    d.classList.remove('bg-amber-400');
                    d.classList.add('bg-white/50');
                }
            });
        }

        function changeSlide(step) {
            showSlide(currentSlideIndex + step);
            resetSlideTimer();
        }

        function goToSlide(index) {
            showSlide(index);
            resetSlideTimer();
        }

        function resetSlideTimer() {
            if (slideInterval) clearInterval(slideInterval);
            slideInterval = setInterval(() => {
                showSlide(currentSlideIndex + 1);
            }, 4500);
        }

        window.onload = function() {
            renderAllProductSections();
            resetSlideTimer();
        };
